; Physical interoperability fixture, NOT a production connection scheduler.
; Generated header supplies shared TLS, QUIC, UDP, board entropy and test CA.
; Runs TLS arithmetic through the shared cooperative worker; UDP stays in
; the foreground dispatcher, outside the worker context.
; No Retry, migration, recovery timer, application frames or HTTP/3 requests.
; The final protected 1-RTT packet is authenticated, but its frames are NOT
; interpreted. TLS completion is also checked independently by the server.

EndStructure

Procedure.i Service()
 Define t.i,rc.i,cost.i
 pet() : t=clock()
 If lastPump<>0 And t-lastPump>maxPumpGap : maxPumpGap=t-lastPump : EndIf
 lastPump=t : rc=UdpPayloadService() : cost=clock()-t
 If cost>maxService : maxService=cost : EndIf
 If clock()-started>20000000 : ProcedureReturn -9301 : EndIf
 ProcedureReturn rc
EndProcedure
Procedure.i Pump()
 Define rc.i
 rc=Service()
 If rc<0 : ProcedureReturn rc : EndIf
 ProcedureReturn rc
EndProcedure
Procedure.i WaitTx()
 Define rc.i
 Repeat
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  rc=UdpPayloadStatus(handle,@status) : If rc<>0 : ProcedureReturn rc : EndIf
  If status\txState=#UDP_TX_SENT : ProcedureReturn 0 : EndIf
  If status\txState=#UDP_TX_FAILED : ProcedureReturn status\txError : EndIf
 ForEver
EndProcedure

Procedure.i SendSpace(*s.LiveSpace)
 Define rc.i,i.i,n.i,head.i,room.i,pn.i,token.i,cursor.i,found.i,view.QuicCryptoSlice,meta.i,copy.i
 ; Locate one bounded ready range. The test's output messages fit one packet.
 cursor=*s\tx\base
 Repeat
  found=QuicCryptoTxFind(@*s\tx,cursor,1024,@view)
  If found<0 : ProcedureReturn found : EndIf
  If found=1 : Break : EndIf
  cursor=view\offset
 Until cursor>=*s\tx\tail
 If found=0 And *s\ack\pending=0 : ProcedureReturn 0 : EndIf
 pn=QuicHandshakeSentReserve(@*s\history) : If pn<0 : ProcedureReturn pn : EndIf
 spec\kind=*s\level : spec\source=@clientId[0] : spec\sourceBytes=8
 If peerBytes=0 : spec\destination=@originalId[0] : spec\destinationBytes=8
 Else : spec\destination=@peerId[0] : spec\destinationBytes=peerBytes : EndIf
 spec\number=pn : spec\numberBytes=4
 head=21+spec\destinationBytes+Bool(*s\level=0)
 spec\cipherBytes=1200-head
 n=QuicLongPacketWrite(@spec,@output[0],1472) : If n<>head : ProcedureReturn -9303 : EndIf
 For i=head To 1199 : output[i]=0 : Next
 room=1200-head-16
 If *s\ack\pending<>0
  rc=QuicHandshakeAckWrite(@*s\ack,@output[n],room,@token)
  If rc<0 : ProcedureReturn rc : EndIf
  n=n+rc : room=room-rc
 EndIf
 If found=1
  copy=view\bytes : If copy>room-24 : copy=room-24 : EndIf
  If copy<=0 : ProcedureReturn -9304 : EndIf
  rc=QuicCryptoTxFrame(@*s\tx,view\offset,copy,@output[n],room)
  If rc<0 : ProcedureReturn rc : EndIf
 EndIf
 rc=QuicPacketJobBegin(@sendJob,@*s\writeKeys,@output[0],1200,0,pn,1)
 If rc<>0 : ProcedureReturn rc : EndIf
 Repeat
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  rc=QuicPacketJobStep(@sendJob)
 Until rc<>0
 If rc<>1 : ProcedureReturn rc : EndIf
 rc=UdpPayloadSendTo(handle,@ep,@output[0],1200) : If rc<1 : ProcedureReturn rc : EndIf
 meta=*s\metadata+pn*16 : PokeI(meta,view\offset) : PokeI(meta+8,copy)
 rc=QuicHandshakeSentCommit(@*s\history,pn,1200,clock(),Bool(copy>0),1,meta) : If rc<>0 : ProcedureReturn rc : EndIf
 If copy>0 : rc=QuicCryptoTxSent(@*s\tx,view\offset,copy) : If rc<>0 : ProcedureReturn rc : EndIf : EndIf
 rc=WaitTx() : If rc<>0 : ProcedureReturn rc : EndIf
 If token<>0
  rc=QuicHandshakeAckSent(@*s\ack,token) : If rc<>1 : ProcedureReturn -9305 : EndIf
 EndIf
 sent=sent+1
 If *s\level=2 And initialDiscarded=0
  QuicKeysWipe(@initial\readKeys) : QuicKeysWipe(@initial\writeKeys)
  rc=QuicHandshakeSentDiscard(@initial\history) : If rc<>0 : ProcedureReturn rc : EndIf
  initialDiscarded=1
 EndIf
 ProcedureReturn 0
EndProcedure
Procedure.i HandleAck(*s.LiveSpace)
 Define rc.i,p.i,n.i
 rc=QuicHandshakeSentAckBegin(@ackJob,@*s\history,*s\level,@receive\process\frame)
 If rc<>0 : ProcedureReturn rc : EndIf
 Repeat
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  rc=QuicHandshakeSentAckStep(@ackJob)
  If rc=1
   ackEvents=ackEvents+1 : n=PeekI(ackJob\cookie+8)
   If n>0
    p=QuicCryptoTxAck(@*s\tx,PeekI(ackJob\cookie),n) : If p<>0 : ProcedureReturn p : EndIf
    p=QuicCryptoTxReclaim(@*s\tx,4096) : If p<0 : ProcedureReturn p : EndIf
   EndIf
  EndIf
 Until rc<>0 And rc<>1
 If rc<>2 : ProcedureReturn rc : EndIf
 ProcedureReturn QuicHandshakeReceiveAccept(@receive)
EndProcedure
Procedure.i FeedTls(*s.LiveSpace)
 Define rc.i,p.i,stepStarted.i,stepCost.i
 Repeat
  p=Pump() : If p<0 : ProcedureReturn p : EndIf
  stepStarted=clock() : rc=DoTls(3,*s) : stepCost=clock()-stepStarted
  If stepCost>maxTlsStep : maxTlsStep=stepCost : EndIf
 Until rc<=0
 If rc<0 : ProcedureReturn rc : EndIf
 If hsReady=0 And Tls13State>=#TLS13_ST_WAIT_EE And Tls13State<=#TLS13_ST_CONNECTED
  rc=KeyDone(QuicTrafficKeysBegin(@keyjob,@handshake\readKeys,@secTlsSHs[0],32)) : If rc<>0 : ProcedureReturn rc : EndIf
  rc=KeyDone(QuicTrafficKeysBegin(@keyjob,@handshake\writeKeys,@secTlsCHs[0],32)) : If rc<>0 : ProcedureReturn rc : EndIf
  hsReady=1
 EndIf
 If paramsReady=0 And PeekI(@tlsConfig[0]+40)<>0
  rc=QuicParametersBegin(@params,@peerParameters[0],PeekI(@tlsConfig[0]+32),1,@parameterEntries[0],64)
  If rc<>0 : ProcedureReturn rc : EndIf
  Repeat : rc=QuicParametersStep(@params) : Until rc<>0
  If rc<>1 : ProcedureReturn rc : EndIf
  rc=QuicParametersCheckIds(@params,@peerId[0],peerBytes,@originalId[0],8,0,0,0)
  If rc<>1 : ProcedureReturn rc : EndIf
  paramsReady=1
 EndIf
 If Tls13State=#TLS13_ST_SEND_FINISHED
  If paramsReady<>1 : ProcedureReturn -9306 : EndIf
  rc=KeyDone(QuicTrafficKeysBegin(@keyjob,@appRead,@secTlsSAp[0],32)) : If rc<>0 : ProcedureReturn rc : EndIf
  rc=MakeOutput(@handshake) : If rc<>0 : ProcedureReturn rc : EndIf
 EndIf
 ProcedureReturn 0
EndProcedure
Procedure.i PacketOne(p.i,bytes.i)
 Define rc.i,i.i,*s.LiveSpace
 rc=QuicPacketRead(@header,p,bytes,8) : If rc<>1 : ProcedureReturn -9307 : EndIf
 If header\destinationBytes<>8 : ProcedureReturn -9308 : EndIf
 For i=0 To 7 : If PeekA(p+header\destinationOffset+i)<>clientId[i] : ProcedureReturn -9309 : EndIf : Next
 If header\kind=4
  If Tls13State<>#TLS13_ST_CONNECTED : ProcedureReturn -9310 : EndIf
  rc=QuicPacketJobBegin(@sendJob,@appRead,p,header\packetBytes,8,-1,0) : If rc<>0 : ProcedureReturn rc : EndIf
  Repeat
   rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
   rc=QuicPacketJobStep(@sendJob)
  Until rc<>0
  If rc<>1 : ProcedureReturn rc : EndIf
  appAuthenticated=1 : ProcedureReturn 0
 EndIf
 If header\version<>1 Or (header\kind<>0 And header\kind<>2) : ProcedureReturn -9311 : EndIf
 If header\kind=0 And initialDiscarded : ProcedureReturn 0 : EndIf
 If header\sourceBytes<1 Or header\sourceBytes>20 : ProcedureReturn -9312 : EndIf
 If peerBytes<>0
  If header\sourceBytes<>peerBytes : ProcedureReturn -9313 : EndIf
  For i=0 To peerBytes-1 : If PeekA(p+header\sourceOffset+i)<>peerId[i] : ProcedureReturn -9314 : EndIf : Next
 EndIf
 If header\kind=0 : *s=@initial
 Else
  If hsReady=0 : ProcedureReturn -9315 : EndIf
  *s=@handshake
 EndIf
 rc=QuicHandshakeReceiveBegin(@receive,@*s\readKeys,p,header\packetBytes,*s\level,*s\largest,@*s\rx,@*s\ack)
 If rc<>0 : ProcedureReturn rc : EndIf
 Repeat
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  rc=QuicHandshakeReceiveStep(@receive)
  If rc=2 : rc=HandleAck(*s) : If rc=0 : rc=1 : EndIf : EndIf
 Until rc<>1
 If rc=6 : ProcedureReturn 0 : EndIf
 If rc<>4 : ProcedureReturn rc : EndIf
 If receive\crypto\number>*s\largest : *s\largest=receive\crypto\number : EndIf
 If peerBytes=0
  peerBytes=header\sourceBytes
  For i=0 To peerBytes-1 : peerId[i]=PeekA(p+header\sourceOffset+i) : Next
 EndIf
 rc=FeedTls(*s) : If rc<>0 : ProcedureReturn rc : EndIf
 ProcedureReturn SendSpace(*s)
EndProcedure
Procedure.i Run()
 Define rc.i,n.i,i.i,p.i,packetBytes.i
 stage=1
 rc=InitSpace(@initial,0,0) : If rc<>0 : ProcedureReturn rc : EndIf
 rc=InitSpace(@handshake,2,1) : If rc<>0 : ProcedureReturn rc : EndIf
 X509ResetAnchors() : X509AddAnchorEC(1,?RootDn,#ROOT_DN_BYTES,23,?RootKey,65)
 Tls13SetHsBuffer(@tlsHs[0],32768) : Tls13SetRecvBuffer(@tlsRecv[0],16645)
 QuicCipherInvalidateHook=@Tls13RecInvalidateCipher
 rc=Tls13Init(?HostName,15,#TEST_UNIX) : If rc<>1 : ProcedureReturn -9316 : EndIf
 stage=2
 rc=DrbgAutoSeed() : If rc<>1 : ProcedureReturn rc : EndIf
 rc=DrbgGenerate(@originalId[0],8) : If rc<>8 : ProcedureReturn -9317 : EndIf
 rc=DrbgGenerate(@clientId[0],8) : If rc<>8 : ProcedureReturn -9318 : EndIf
 rc=DoTls(1,0) : If rc<>1 : ProcedureReturn -9319 : EndIf
 rc=Tls13SetAlpn(?Alpn,3) : If rc<>1 : ProcedureReturn -9320 : EndIf
 For i=0 To #LOCAL_PARAMETER_BYTES-1 : localParameters[i]=PeekA(?LocalTemplate+i) : Next
 For i=0 To 7 : localParameters[i+2]=clientId[i] : Next
 rc=Tls13SetQuicParameters(@tlsConfig[0],@localParameters[0],#LOCAL_PARAMETER_BYTES,@peerParameters[0],4096)
 If rc<>1 : ProcedureReturn -9321 : EndIf
 rc=KeyDone(QuicInitialKeysBegin(@keyjob,@initial\readKeys,@originalId[0],8,1)) : If rc<>0 : ProcedureReturn rc : EndIf
 rc=KeyDone(QuicInitialKeysBegin(@keyjob,@initial\writeKeys,@originalId[0],8,0)) : If rc<>0 : ProcedureReturn rc : EndIf
 stage=3
 rc=MakeOutput(@initial) : If rc<>0 : ProcedureReturn rc : EndIf
 rc=SendSpace(@initial) : If rc<>0 : ProcedureReturn rc : EndIf
 stage=4
 While appAuthenticated=0
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  n=UdpPayloadReceive(handle,@info,@packet[0],1472)
  If n<>#SVC_EAGAIN
   If n<1 : ProcedureReturn n : EndIf
   If info\family<>4 Or info\peerPort<>55442 : ProcedureReturn -9322 : EndIf
   For i=0 To 15 : If PeekA(@info\peerAddress+i)<>PeekA(@ep\address+i) : ProcedureReturn -9323 : EndIf : Next
   received=received+1 : p=0
   While p<n
    rc=QuicPacketRead(@header,@packet[p],n-p,8) : If rc<>1 : ProcedureReturn -9324 : EndIf
    packetBytes=header\packetBytes
    rc=PacketOne(@packet[p],packetBytes) : If rc<>0 : ProcedureReturn rc : EndIf
    p=p+packetBytes
   Wend
  EndIf
 Wend
 stage=5 : ProcedureReturn 0
EndProcedure

Procedure ForeignContext()
 Tls13State=901 : Tls13Err=902 : TlsNameLen=42 : TlsQuicConfig=0
 TlsClientRandom[0]=167 : TlsSessionId[0]=89 : TlsGroup=903
EndProcedure
Procedure CheckContext()
 Define i.i,n.i
 n=Tls13ContextSave(@afterSnapshot[0],#TLS13_CONTEXT_SIZE)
 For i=0 To #TLS13_CONTEXT_SIZE-1
  If beforeSnapshot[i]<>afterSnapshot[i] : contextFault=1 : EndIf
  ; White-box check: the temporary foreground secret snapshot is wiped.
  If PeekA(@tlsctxw_caller[0]+i)<>0 : contextFault=2 : EndIf
 Next
 contextChecks=contextChecks+1
EndProcedure
Procedure.i BeginOperation(operation.i,space.i)
 Define *s.LiveSpace
 *s=space

 If operation=2 : ProcedureReturn QuicTlsOperationBegin(@liveOp,@ownerSnapshot[0],#TLS13_CONTEXT_SIZE,2,@*s\tx,@tlsOutput,@tlsBytes[0],2048,20000000) : EndIf
 ProcedureReturn QuicTlsOperationBegin(@liveOp,@ownerSnapshot[0],#TLS13_CONTEXT_SIZE,3,@*s\rx,0,0,0,20000000)
EndProcedure
Procedure OperationProbe(kind.i)
 Define i.i,rc.i,n.i,ticks.i,t.i
 For i=0 To #TLS13_CONTEXT_SIZE-1 : probeSnapshot[i]=ownerSnapshot[i] : Next
 ForeignContext() : n=Tls13ContextSave(@beforeSnapshot[0],#TLS13_CONTEXT_SIZE)
 Select kind
  Case 1
   rc=QuicTlsOperationBegin(@deadlineOp,@probeSnapshot[0],#TLS13_CONTEXT_SIZE,1,0,0,0,0,1)
   If rc<>1 : operationFault=1 : ProcedureReturn : EndIf
   t=HwMicros() : While HwMicros()-t<3 : Wend
   rc=QuicTlsOperationStep(@deadlineOp) : CheckContext() : extraContextChecks=extraContextChecks+1
   If rc<>#QUIC_OP_TIMEOUT Or deadlineOp\phase<>-1 Or TlsWorkerBusy()<>0 : operationFault=2 : EndIf
   If QuicTlsOperationBegin(@deadlineOp,@probeSnapshot[0],#TLS13_CONTEXT_SIZE,1,0,0,0,0,1000)<>#QUIC_OP_STATE : operationFault=3 : EndIf
  Case 2
   rc=QuicCryptoTxInit(@probeTx,0,@probeData[0],4096,@probeBits[0],1024)
   If rc<>0 : operationFault=4 : ProcedureReturn : EndIf
   rc=QuicTlsOperationBegin(@outputOp,@probeSnapshot[0],#TLS13_CONTEXT_SIZE,2,@probeTx,@probeOut,@probeOutput[0],2048,20000000)
   If rc<>1 : operationFault=5 : ProcedureReturn : EndIf
   ; Deterministic cancellation checkpoint only: restore the production slice
   ; immediately afterward. Live handshake timing never uses this test setting.
   ticks=tlsjob_sliceTicks : tlsjob_sliceTicks=1
   rc=QuicTlsOperationStep(@outputOp) : tlsjob_sliceTicks=ticks
   CheckContext() : extraContextChecks=extraContextChecks+1
   If rc<>0 Or probeOut\phase<>3 : operationFault=6 : EndIf
   rc=QuicTlsOperationCancel(@outputOp) : CheckContext() : extraContextChecks=extraContextChecks+1
   If rc<>#QUIC_OP_CANCELLED Or probeOut\phase<>-1 Or probeTx\phase<>3 Or TlsWorkerBusy()<>0 : operationFault=7 : EndIf
   If QuicCryptoTxAppend(@probeTx,@probeOutput[0],1)<>#QUIC_TX_STATE : operationFault=8 : EndIf
  Case 3
   rc=QuicCryptoRxInit(@probeRx,0,@probeData[0],4096,@probeBits[0],512)
   If rc<>0 : operationFault=9 : ProcedureReturn : EndIf
   rc=QuicTlsOperationBegin(@inputOp,@probeSnapshot[0],#TLS13_CONTEXT_SIZE,3,@probeRx,0,0,0,20000000)
   If rc<>1 : operationFault=10 : ProcedureReturn : EndIf
   rc=QuicTlsOperationCancel(@inputOp) : CheckContext() : extraContextChecks=extraContextChecks+1
   If rc<>#QUIC_OP_CANCELLED Or probeRx\phase<>3 Or probeRx\offered<>0 Or TlsWorkerBusy()<>0 : operationFault=11 : EndIf
 EndSelect
 For i=0 To #TLS13_CONTEXT_SIZE-1 : If probeSnapshot[i]<>0 : operationFault=12 : EndIf : Next
 n=Tls13ContextRestore(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
 operationProbes=operationProbes+1
EndProcedure
Procedure.i DoTls(operation.i,space.i)
 Define rc.i,t.i,cost.i,n.i,i.i
 liveOperation=operation : liveSpace=space
 otherJob\result=773
 refusedJob\snapshot=771 : refusedJob\capacity=772 : refusedJob\worker\result=773
 n=Tls13ContextSave(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
 If TlsContextWorkerBegin(@refusedJob,@refusedJob,#TLS13_CONTEXT_SIZE,@LiveEntry,0)<>#TLSWORK_STATE : contextFault=5 : EndIf
 If operation=1 And cancelChecked=0
  For i=0 To #TLS13_CONTEXT_SIZE-1 : retrySnapshot[i]=ownerSnapshot[i] : Next
  rc=TlsContextWorkerBegin(@liveJob,@ownerSnapshot[0],#TLS13_CONTEXT_SIZE,@LiveEntry,0)
  If rc<>1 : ProcedureReturn rc : EndIf
  ForeignContext() : n=Tls13ContextSave(@beforeSnapshot[0],#TLS13_CONTEXT_SIZE)
  rc=TlsContextWorkerStep(@liveJob) : CheckContext()
  If rc<>0 : contextFault=3 : ProcedureReturn -9326 : EndIf
  rc=TlsContextWorkerCancel(@liveJob) : CheckContext()
  If rc<>1 : ProcedureReturn rc : EndIf
