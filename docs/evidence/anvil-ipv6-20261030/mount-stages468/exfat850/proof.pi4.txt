; Native exFAT acceptance against a writable memory block device. This is a
; diagnostic fixture, not shipped product code and it touches no hardware.
XIncludeFile "RaspberryPi4/Lib/fat.pi4"

#GATE_BLOCKS = 2100
#GATE_BYTES = #GATE_BLOCKS * 512
Global Dim gateDisk.a[#GATE_BYTES]
Global Dim gateWrite.a[1400]
Global Dim gateRead.a[1400]
Global Dim gateName.a[1024]
Global Dim gateBig.a[4608]
Global gateFailLba.i=-1
Global gateFlushes.i
Global gateWrites.i
Global gateReads.i

; The block-range seam: count blocks per call. gateReads and gateWrites count
; CALLS, which is what a USB command or an SD transfer costs.
Procedure.i GateReader(lba.i, count.i, *dst)
  If lba < 0 Or count < 1 Or lba + count > #GATE_BLOCKS
    ProcedureReturn 0
  EndIf
  If gateFailLba>=lba And gateFailLba<lba+count : ProcedureReturn 0 : EndIf
  gateReads = gateReads + 1
  exfat_Copy(*dst, @gateDisk[0] + lba * 512, count * 512)
  ProcedureReturn 1
EndProcedure

Procedure.i GateWriter(lba.i, count.i, *src)
  If lba < 0 Or count < 1 Or lba + count > #GATE_BLOCKS
    ProcedureReturn 0
  EndIf
  gateWrites = gateWrites + 1
  exfat_Copy(@gateDisk[0] + lba * 512, *src, count * 512)
  ProcedureReturn 1
EndProcedure

Procedure.i GateFlusher()
  gateFlushes = gateFlushes + 1
  ProcedureReturn 1
EndProcedure

Procedure.i GateBootChecksum(base.i)
  Define sum.i
  Define s.i
  Define i.i
  Define b.i
  sum = 0
  s = 0
  While s < 11
    i = 0
    While i < 512
      If s <> 0 Or (i <> 106 And i <> 107 And i <> 112)
        b = gateDisk[(base + s) * 512 + i]
        sum = (exfat_Ror32(sum) + b) & $FFFFFFFF
      EndIf
      i = i + 1
    Wend
    s = s + 1
  Wend
  ProcedureReturn sum
EndProcedure

Procedure.i GateCrc32(offset.i, length.i)
  Define crc.i
  Define i.i
  crc = $FFFFFFFF
  i = 0
  While i < length
    crc = exfat_Crc32Update(crc, gateDisk[offset + i])
    i = i + 1
  Wend
  ProcedureReturn (~crc) & $FFFFFFFF
EndProcedure

Procedure GateMoveToGpt()
  Define s.i
  Define i.i
  Define sum.i
  Define crc.i
  ; Move the complete 153-sector volume before replacing its MBR with GPT.
  s = 152
  While s >= 0
    exfat_Copy(@gateDisk[0] + (40 + s) * 512, @gateDisk[0] + (1 + s) * 512, 512)
    s = s - 1
  Wend
  exfat_Zero(@gateDisk[0], 40 * 512)
  ; Protective MBR.
  gateDisk[$1BE + 4] = $EE
  exfat_Put32(@gateDisk[0], $1BE + 8, 1)
  exfat_Put32(@gateDisk[0], $1BE + 12, 2099)
  gateDisk[510] = $55 : gateDisk[511] = $AA
  ; One Microsoft Basic Data entry in the standard 16 KiB entry array.
  exfat_Copy(@gateDisk[0] + 2 * 512, ?gateGptBasic, 16)
  i = 0
  While i < 16
    gateDisk[2 * 512 + 16 + i] = i + 1
    i = i + 1
  Wend
  exfat_Put64(@gateDisk[0] + 2 * 512, 32, 40)
  exfat_Put64(@gateDisk[0] + 2 * 512, 40, 2087)
  crc = GateCrc32(2 * 512, 128 * 128)
  ; Primary GPT header.
  exfat_Copy(@gateDisk[0] + 512, ?gateGptSignature, 8)
  exfat_Put32(@gateDisk[0] + 512, 8, $00010000)
  exfat_Put32(@gateDisk[0] + 512, 12, 92)
  exfat_Put64(@gateDisk[0] + 512, 24, 1)
  exfat_Put64(@gateDisk[0] + 512, 32, 2099)
  exfat_Put64(@gateDisk[0] + 512, 40, 34)
  exfat_Put64(@gateDisk[0] + 512, 48, 2098)
  exfat_Put64(@gateDisk[0] + 512, 72, 2)
  exfat_Put32(@gateDisk[0] + 512, 80, 128)
  exfat_Put32(@gateDisk[0] + 512, 84, 128)
  exfat_Put32(@gateDisk[0] + 512, 88, crc)
  crc = GateCrc32(512, 92)
  exfat_Put32(@gateDisk[0] + 512, 16, crc)
  ; PartitionOffset changes with the move and is protected by both boot sums.
  exfat_Put64(@gateDisk[0] + 40 * 512, 64, 40)
  exfat_Put64(@gateDisk[0] + 52 * 512, 64, 40)
  sum = GateBootChecksum(40)
  i = 0
  While i < 512
    exfat_Put32(@gateDisk[0] + 51 * 512, i, sum)
    i = i + 4
  Wend
  sum = GateBootChecksum(52)
  i = 0
  While i < 512
    exfat_Put32(@gateDisk[0] + 63 * 512, i, sum)
    i = i + 4
  Wend
EndProcedure

Procedure GateFormat()
  Define i.i
  Define s.i
  Define sum.i
  Define tableSum.i
  Define off.i
  Define cp.i
  Define base.i
  Define root.i
  exfat_Zero(@gateDisk[0], #GATE_BYTES)
  ; MBR, one type-07 partition at physical block 1, 200 blocks long.
  gateDisk[$1BE + 4] = 7
  exfat_Put32(@gateDisk[0], $1BE + 8, 1)
  exfat_Put32(@gateDisk[0], $1BE + 12, 2048)
  gateDisk[510] = $55 : gateDisk[511] = $AA

  ; Main and backup boot regions. Logical sectors are 512-byte blocks.
  base = 1
  For s = 0 To 23
    gateDisk[(base + s) * 512 + 510] = $55
    gateDisk[(base + s) * 512 + 511] = $AA
  Next
  For s = 0 To 12 Step 12
    off = (base + s) * 512
    gateDisk[off] = $EB : gateDisk[off + 1] = $76 : gateDisk[off + 2] = $90
    exfat_Copy(@gateDisk[0] + off + 3, ?gateSignature, 8)
    exfat_Put64(@gateDisk[0] + off, 64, 1)
    exfat_Put64(@gateDisk[0] + off, 72, 2048)
    exfat_Put32(@gateDisk[0] + off, 80, 24)
    exfat_Put32(@gateDisk[0] + off, 84, 2)
    exfat_Put32(@gateDisk[0] + off, 88, 26)
    exfat_Put32(@gateDisk[0] + off, 92, 128)
    exfat_Put32(@gateDisk[0] + off, 96, 4)
    exfat_Put32(@gateDisk[0] + off, 100, $12345678)
    exfat_Put16(@gateDisk[0] + off, 104, $0100)
    exfat_Put16(@gateDisk[0] + off, 106, 0)
    gateDisk[off + 108] = 9
    gateDisk[off + 109] = 0
    gateDisk[off + 110] = 1
    gateDisk[off + 112] = 3
  Next

  ; FAT: bitmap, up-case table and root are one-cluster chains.
  off = (base + 24) * 512
  exfat_Put32(@gateDisk[0] + off, 2 * 4, $FFFFFFFF)
  exfat_Put32(@gateDisk[0] + off, 3 * 4, $FFFFFFFF)
  exfat_Put32(@gateDisk[0] + off, 4 * 4, $FFFFFFFF)

  ; Allocation bitmap in cluster 2: clusters 2, 3 and 4 are allocated.
  off = (base + 26) * 512
  gateDisk[off] = 7

  ; Compact up-case table in cluster 3. ASCII lower-case maps to upper-case;
  ; all other UTF-16 code units are identity-mapped with the compressed run.
  off = (base + 27) * 512
  cp = 0
  While cp < 97
    exfat_Put16(@gateDisk[0] + off, cp * 2, cp)
    cp = cp + 1
  Wend
  i = 0
  While i < 26
    exfat_Put16(@gateDisk[0] + off, 194 + i * 2, 65 + i)
    i = i + 1
  Wend
  exfat_Put16(@gateDisk[0] + off, 246, $FFFF)
  exfat_Put16(@gateDisk[0] + off, 248, 65413)
  tableSum = 0
  i = 0
  While i < 250
    tableSum = (exfat_Ror32(tableSum) + gateDisk[off + i]) & $FFFFFFFF
    i = i + 1
  Wend

  ; Root directory in cluster 4: allocation bitmap, up-case table, end.
  root = (base + 28) * 512
  gateDisk[root] = $81
  gateDisk[root + 1] = 0
  exfat_Put32(@gateDisk[0] + root, 20, 2)
  exfat_Put64(@gateDisk[0] + root, 24, 16)
  gateDisk[root + 32] = $82
  exfat_Put32(@gateDisk[0] + root + 32, 4, tableSum)
  exfat_Put32(@gateDisk[0] + root + 32, 20, 3)
  exfat_Put64(@gateDisk[0] + root + 32, 24, 250)
  gateDisk[root + 64] = 0

  ; Boot checksum sectors, one in each region.
  sum = GateBootChecksum(base)
  i = 0
  While i < 512
    exfat_Put32(@gateDisk[0] + (base + 11) * 512, i, sum)
    i = i + 4
  Wend
  ; Backup is byte-identical to main after its own checksum sector.
  exfat_Copy(@gateDisk[0] + (base + 12) * 512, @gateDisk[0] + base * 512, 12 * 512)
EndProcedure

Procedure.i GateBytes(count.i, seed.i)
  Define i.i
  i = 0
  While i < count
    If gateRead[i] <> ((i * 37 + seed) & $FF)
      ProcedureReturn 0
    EndIf
    i = i + 1
  Wend
  ProcedureReturn 1
EndProcedure

; ----------------------------------------------------------------------
;  THE FILESYSTEM FACADE: partitions named in the path, and the two read
;  paths a real medium exposed. The same volume is moved to MBR slot 2 with
;  slot 1 left empty, so a path that reaches it proves the prefix works and
;  a path without one proves it did NOT silently reach partition 2.
; ----------------------------------------------------------------------
Procedure.i GateBigBytes(count.i, seed.i)
  Define i.i
  i = 0
  While i < count
    If gateBig[i] <> ((i * 13 + seed) & $FF)
      ProcedureReturn 0
    EndIf
    i = i + 1
  Wend
  ProcedureReturn 1
EndProcedure

Procedure.i GateFacade(free0.i)
  Define i.i
  Define n.i
  Define seen.i
  Define writes.i
  exfat_Copy(@gateDisk[0] + $1BE + 16, @gateDisk[0] + $1BE, 16)
  exfat_Zero(@gateDisk[0] + $1BE, 16)
  FsSetRangeReader(@GateReader)
  FsSetRangeWriter(@GateWriter)
  FsSetBlockFlusher(@GateFlusher)
  ; A partition digit outside 1..4 is the facade's refusal, not a driver's.
  If FsSelectPartition(5) <> 0 Or FsErrorKind() <> #FS_FACADE Or FsLastError() <> #FS_E_PARTITION : ProcedureReturn 100 : EndIf
  If FsOpen(?fsBad5) <> 0 Or FsErrorKind() <> #FS_FACADE Or FsLastError() <> #FS_E_PARTITION : ProcedureReturn 101 : EndIf
  If FsMkdir(?fsDataDir) = 0 : ProcedureReturn 102 : EndIf
  If FsType() <> #FS_EXFAT Or FsPartition() <> 2 : ProcedureReturn 103 : EndIf
  ; a.bin gets one cluster, b.bin the next, then a.bin grows past b.bin:
  ; a FAT-chained file of nine one-sector clusters.
  i = 0
  While i < 4608
    gateBig[i] = (i * 13 + 5) & $FF
    i = i + 1
  Wend
  If FsCreate(?fsA) = 0 Or FsWrite(@gateBig[0], 512) <> 512 : ProcedureReturn 104 : EndIf
  FsClose()
  If FsCreate(?fsB) = 0 Or FsWrite(@gateBig[0], 512) <> 512 : ProcedureReturn 105 : EndIf
  FsClose()
  If FsOpen(?fsA) = 0 Or FsSeek(512) = 0 Or FsWrite(@gateBig[0] + 512, 4096) <> 4096 : ProcedureReturn 106 : EndIf
  FsClose()
  If FsFlush() = 0 : ProcedureReturn 107 : EndIf
  ; ONE READ OF THE WHOLE CHAINED FILE COSTS ONE FAT LOOKUP PER CLUSTER.
  ; Resolving each sector from the first cluster costs 0+1+...+8 = 36 FAT
  ; reads on top of the nine data sectors; the cursor costs 8.
  If FsOpen(?fsA) = 0 Or FsSize() <> 4608 : ProcedureReturn 108 : EndIf
  exfat_Zero(@gateBig[0], 4608)
  gateReads = 0
  If FsRead(@gateBig[0], 4608) <> 4608 : ProcedureReturn 109 : EndIf
  If gateReads > 20 : ProcedureReturn 1000 + gateReads : EndIf
  If GateBigBytes(4608, 5) = 0 : ProcedureReturn 110 : EndIf
  FsClose()
  ; No prefix is partition 1, which is empty here. It must NOT reach the
  ; volume on partition 2, and its refusal must not read as "not found" -
  ; save creates a file on "not found".
  If FsOpen(?fsAUnprefixed) <> 0 Or FsPartition() <> 0 : ProcedureReturn 111 : EndIf
  If FsErrorKind() <> #FS_FAT Or FsErrorIsNotFound() <> 0 : ProcedureReturn 112 : EndIf
  If FsRename(?fsB, ?fsBCross) <> 0 Or FsErrorKind() <> #FS_FACADE Or FsLastError() <> #FS_E_CROSS_VOLUME : ProcedureReturn 113 : EndIf
  If FsRename(?fsB, ?fsBRenamed) = 0 : ProcedureReturn 114 : EndIf
  If FsOpen(?fsB) <> 0 Or FsErrorIsNotFound() = 0 : ProcedureReturn 115 : EndIf
  ; The last lookup resolved a FILE. An empty path must not inherit it.
  If FsOpen(?fsBRenamed) = 0 : ProcedureReturn 116 : EndIf
  FsClose()
  writes = gateWrites
  If FsOpen(?fsRoot) <> 0 Or FsErrorKind() <> #FS_EXFAT Or FsLastError() <> #EXFAT_E_PATH : ProcedureReturn 117 : EndIf
  If FsRemove(?fsRootBare) <> 0 Or FsLastError() <> #EXFAT_E_PATH Or gateWrites <> writes : ProcedureReturn 118 : EndIf
  ; The folder lists by long name through the prefix, and so does the root.
  If FsDirOpen(?fsDataDir) = 0 : ProcedureReturn 119 : EndIf
  seen = 0
  n = FsDirNext()
  While n > 0
    seen = seen + 1
    n = FsDirNext()
  Wend
  If n < 0 Or seen <> 2 : ProcedureReturn 120 + seen : EndIf
  If FsDirOpen(?fsRootBare) = 0 : ProcedureReturn 130 : EndIf
  seen = 0
  n = FsDirNext()
  While n > 0
    seen = seen + 1
    n = FsDirNext()
  Wend
  If n < 0 Or seen <> 1 : ProcedureReturn 131 + seen : EndIf
  If FsRemove(?fsA) = 0 Or FsRemove(?fsBRenamed) = 0 Or FsRemove(?fsDataDir) = 0 : ProcedureReturn 140 : EndIf
  If FsFlush() = 0 : ProcedureReturn 141 : EndIf
  If ExFatFreeClusters() <> free0 : ProcedureReturn 142 : EndIf
  ; Clean on the medium, in both boot regions, once flushed.
  If (exfat_U16(@gateDisk[0] + 1 * 512, 106) & 2) <> 0 Or (exfat_U16(@gateDisk[0] + 13 * 512, 106) & 2) <> 0 : ProcedureReturn 143 : EndIf
  If FsUnmount() = 0 : ProcedureReturn 144 : EndIf
  FsSetRangeWriter(0)
  exfat_Copy(@gateDisk[0] + $1BE, @gateDisk[0] + $1BE + 16, 16)
  exfat_Zero(@gateDisk[0] + $1BE + 16, 16)
  ProcedureReturn 0
EndProcedure

Procedure.i GateMain()
  Define i.i
  Define n.i
  Define free0.i
  Define seen.i
  GateFormat()
  ExFatSetRangeReader(@GateReader)
  ; A read-only mount and unmount never mark the volume or call a writer.
  ExFatSetRangeWriter(0)
  ExFatSetBlockFlusher(0)
  If ExFatMount(1) = 0 : ProcedureReturn 2900 + ExFatLastError() : EndIf
  If ExFatUnmount() = 0 : ProcedureReturn 2950 + ExFatLastError() : EndIf
  If gateWrites <> 0 : ProcedureReturn 2999 : EndIf
  ; A volume inherited dirty remains readable, but mutation is refused and
  ; Anvil never clears somebody else's dirty state.
  exfat_Put16(@gateDisk[0] + 1 * 512, 106, 2)
  exfat_Put16(@gateDisk[0] + 13 * 512, 106, 2)
  ExFatSetRangeWriter(@GateWriter)
  If ExFatMount(1) = 0 : ProcedureReturn 30 : EndIf
  gateWrites = 0
  If ExFatMkdir(?nameDirty) <> 0 Or ExFatLastError() <> #EXFAT_E_DIRTY Or gateWrites <> 0 : ProcedureReturn 31 : EndIf
  If ExFatUnmount() = 0 Or gateWrites <> 0 : ProcedureReturn 32 : EndIf
  exfat_Put16(@gateDisk[0] + 1 * 512, 106, 0)
  exfat_Put16(@gateDisk[0] + 13 * 512, 106, 0)
  ExFatSetRangeWriter(@GateWriter)
  ExFatSetBlockFlusher(@GateFlusher)
  If ExFatMount(1) = 0 : ProcedureReturn 1 : EndIf
  free0 = ExFatFreeClusters()
  If free0 <> 125 : ProcedureReturn 2 : EndIf
  If ExFatMkdir(?nameDocs) = 0 : ProcedureReturn 3 : EndIf
  If ExFatMkdir(?nameSub) = 0 : ProcedureReturn 4 : EndIf
  If ExFatCreate(?nameLong) = 0 : ProcedureReturn 5 : EndIf
  i = 0
  While i < 700
    gateWrite[i] = (i * 37 + 11) & $FF
    i = i + 1
  Wend
  If ExFatWrite(@gateWrite[0], 700) <> 700 : ProcedureReturn 6 : EndIf
  ExFatClose()
  If ExFatFlush() = 0 Or gateFlushes < 2 : ProcedureReturn 7 : EndIf
  If ExFatUnmount() = 0 : ProcedureReturn 8 : EndIf
  If ExFatMount(1) = 0 : ProcedureReturn 9 : EndIf
  ; ASCII case folding comes from the on-volume up-case table.
  If ExFatOpen(?nameLongFolded) = 0 : ProcedureReturn 10 : EndIf
  n = ExFatRead(@gateRead[0], 700)
  If n <> 700 Or GateBytes(700, 11) = 0 : ProcedureReturn 11 : EndIf
  If ExFatSeek(700) = 0 : ProcedureReturn 12 : EndIf
  i = 0
  While i < 600
    gateWrite[i] = (i * 37 + 29) & $FF
    i = i + 1
  Wend
  If ExFatWrite(@gateWrite[0], 600) <> 600 : ProcedureReturn 13 : EndIf
  If ExFatTruncate(300) = 0 Or ExFatSize() <> 300 : ProcedureReturn 14 : EndIf
  ExFatClose()
  If ExFatRename(?nameLong, ?nameUnicode) = 0 : ProcedureReturn 15 : EndIf
  If ExFatMkdir(?nameChild) = 0 : ProcedureReturn 35 : EndIf
  ; A directory can never be moved beneath itself, including through a
  ; deeper descendant rather than only its immediate child.
  If ExFatRename(?nameDocs, ?nameCycle) <> 0 Or ExFatLastError() <> #EXFAT_E_PATH : ProcedureReturn 36 : EndIf
  If ExFatRemove(?nameChild) = 0 : ProcedureReturn 37 : EndIf
  ; Force the subdirectory past its original one-cluster entry capacity.
  If ExFatCreate(?nameGrow1) = 0 : ProcedureReturn 38 : EndIf : ExFatClose()
  If ExFatCreate(?nameGrow2) = 0 : ProcedureReturn 39 : EndIf : ExFatClose()
  If ExFatCreate(?nameGrow3) = 0 : ProcedureReturn 40 : EndIf : ExFatClose()
  If ExFatCreate(?nameGrow4) = 0 : ProcedureReturn 41 : EndIf : ExFatClose()
  If ExFatOpen(?nameUnicodeFolded) = 0 : ProcedureReturn 16 : EndIf
  If ExFatRead(@gateRead[0], 400) <> 300 Or GateBytes(300, 11) = 0 : ProcedureReturn 17 : EndIf
  ExFatClose()
  If ExFatDirOpen(?nameSub) = 0 : ProcedureReturn 18 : EndIf
  seen = 0
  n = ExFatDirNext()
  While n > 0
    n = ExFatDirNameUtf8(@gateName[0], 1024)
    If n > 0
      seen = seen + 1
    EndIf
    n = ExFatDirNext()
  Wend
  If n < 0 : ProcedureReturn 2000 + seen * 100 + ExFatLastError() : EndIf
  If seen <> 5 : ProcedureReturn 1900 + seen : EndIf
  If ExFatRemove(?nameGrow1) = 0 : ProcedureReturn 42 : EndIf
  If ExFatRemove(?nameGrow2) = 0 : ProcedureReturn 43 : EndIf
  If ExFatRemove(?nameGrow3) = 0 : ProcedureReturn 44 : EndIf
  If ExFatRemove(?nameGrow4) = 0 : ProcedureReturn 45 : EndIf
  If ExFatRemove(?nameUnicode) = 0 : ProcedureReturn 20 : EndIf
  If ExFatRemove(?nameSub) = 0 : ProcedureReturn 21 : EndIf
  If ExFatRemove(?nameDocs) = 0 : ProcedureReturn 22 : EndIf
  If ExFatFreeClusters() <> free0 : ProcedureReturn 23 : EndIf
  If ExFatFlush() = 0 Or ExFatUnmount() = 0 : ProcedureReturn 24 : EndIf
  If ExFatMount(1) = 0 : ProcedureReturn 25 : EndIf
  If ExFatOpen(?nameUnicode) <> 0 Or ExFatLastError() <> #EXFAT_E_NOT_FOUND : ProcedureReturn 26 : EndIf
  If ExFatUnmount() = 0 : ProcedureReturn 27 : EndIf

  n = GateFacade(free0)
  If n <> 0 : ProcedureReturn n : EndIf

  ; The same volume is discovered through a CRC-protected GPT, not only MBR.
  GateMoveToGpt()
  If ExFatMount(1) = 0 Or ExFatUnmount() = 0 : ProcedureReturn 33 : EndIf
  gateDisk[2 * 512 + 128] = 1
  If ExFatMount(1) <> 0 Or ExFatLastError() <> #EXFAT_E_PARTITION : ProcedureReturn 34 : EndIf
  gateDisk[2 * 512 + 128] = 0

  ; A damaged copy is refused before any file metadata is trusted.
  gateDisk[(40 + 2) * 512 + 17] = gateDisk[(40 + 2) * 512 + 17] ! 1
  If ExFatMount(1) <> 0 Or ExFatLastError() <> #EXFAT_E_BOOT_CHECKSUM : ProcedureReturn 28 : EndIf
  ProcedureReturn 0
EndProcedure

Procedure.i OldChecksum()
  Define sum.i
  Define off.i
  Define b.i
  sum = 0
  off = 0
  While off < exfat_upcaseLength
    b = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off)
    If b < 0
      ProcedureReturn -1
    EndIf
    sum = (exfat_Ror32(sum) + b) & $FFFFFFFF
    off = off + 1
  Wend
  ProcedureReturn sum
EndProcedure


Procedure.i OldValidate()
  Define cp.i
  Define off.i
  Define lo.i
  Define hi.i
  Define word.i
  Define skip.i
  cp = 0
  off = 0
  While off + 1 < exfat_upcaseLength And cp < 65536
    lo = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off)
    hi = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off + 1)
    If lo < 0 Or hi < 0
      ProcedureReturn 0
    EndIf
    word = lo | (hi << 8)
    off = off + 2
    If word = $FFFF
      If off + 1 >= exfat_upcaseLength
        If cp = $FFFF
          cp = cp + 1
          Break
        EndIf
        ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
      EndIf
      lo = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off)
      hi = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off + 1)
      If lo < 0 Or hi < 0
        ProcedureReturn 0
      EndIf
      skip = lo | (hi << 8)
      off = off + 2
      If skip = 0 Or cp + skip > 65536
        ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
      EndIf
      cp = cp + skip
    Else
      cp = cp + 1
    EndIf
  Wend
  If cp <> 65536 Or off <> exfat_upcaseLength
    ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
  EndIf
  ProcedureReturn 1
EndProcedure
Procedure.i OldBuildFast()
  Define cp.i
  Define off.i
  Define word.i
  Define skip.i
  Define lo.i
  Define hi.i
  Define endCp.i
  cp = 0
  off = 0
  While cp < 256 And off + 1 < exfat_upcaseLength
    lo = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off)
    hi = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off + 1)
    If lo < 0 Or hi < 0
      ProcedureReturn 0
    EndIf
    word = lo | (hi << 8)
    off = off + 2
    If word = $FFFF
      If off + 1 >= exfat_upcaseLength
        ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
      EndIf
      lo = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off)
      hi = exfat_StreamByte(exfat_upcaseCluster, 0, exfat_upcaseLength, off + 1)
      If lo < 0 Or hi < 0
        ProcedureReturn 0
      EndIf
      skip = lo | (hi << 8)
      off = off + 2
      If skip = 0 Or cp + skip > 65536
        ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
      EndIf
      endCp = cp + skip
      If endCp > 256
        endCp = 256
      EndIf
      While cp < endCp
        exfat_upcaseFast[cp] = cp
        cp = cp + 1
      Wend
    Else
      exfat_upcaseFast[cp] = word
      cp = cp + 1
    EndIf
  Wend
  If cp <> 256
    ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
  EndIf
  ; Microsoft defines mandatory mappings for the first 128 code points.
  cp = 0
  While cp < 128
    If cp >= 97 And cp <= 122
      If exfat_upcaseFast[cp] <> cp - 32
        ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
      EndIf
    ElseIf exfat_upcaseFast[cp] <> cp
      ProcedureReturn exfat_Fail(#EXFAT_E_UPCASE)
    EndIf
    cp = cp + 1
  Wend
  exfat_upcaseFastReady = 1
  ProcedureReturn 1
EndProcedure
