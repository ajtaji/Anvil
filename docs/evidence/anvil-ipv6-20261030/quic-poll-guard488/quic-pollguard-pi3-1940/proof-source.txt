; Physical interoperability fixture, NOT a production connection scheduler.
; Generated header supplies shared TLS, QUIC, UDP, board entropy and test CA.
; Runs TLS arithmetic through the shared cooperative worker; UDP stays in
; the foreground dispatcher, outside the worker context.
; No Retry, migration, complete recovery scheduler or HTTP/3 requests.
; The first protected 1-RTT packet is authenticated and its frame syntax is
; decoded. HANDSHAKE_DONE and NEW_CONNECTION_ID effects run; ACK transmission
; now produces an application ACK/PING after all supported effects succeed.
; Other frame kinds are refused without committing an application receipt.
; TLS completion is also checked independently by the server.

Procedure.i Service()
 Define t.i,rc.i,cost.i
 pet() : t=clock()
 If lastPump<>0 And t-lastPump>maxPumpGap : maxPumpGap=t-lastPump : EndIf
 lastPump=t : rc=UdpPayloadService() : cost=clock()-t
 If cost>maxService : maxService=cost : EndIf
 If rc<0 : ProcedureReturn rc : EndIf
 If connection\phase=1
  rc=QuicConnectionCheckDeadline(@connection,HwMicros())
  If rc<>0 : ProcedureReturn rc : EndIf
 EndIf
 ProcedureReturn rc
EndProcedure

 ProcedureReturn rc
EndProcedure
Procedure.i KeyDone(rc.i)
 Define p.i,n.i,t.i,cost.i
 If rc<>0 : ProcedureReturn rc : EndIf
 n=Tls13ContextSave(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
 rc=QuicTlsOperationBegin(@connection\tlsOp,@ownerSnapshot[0],#TLS13_CONTEXT_SIZE,5,@connection\keyjob,connection\keyjob\keys,0,0,20000000)
 If rc<>1 : ProcedureReturn rc : EndIf
 Repeat
  TlsSliceBefore()
  t=HwMicros() : rc=QuicTlsOperationStep(@connection\tlsOp) : cost=HwMicros()-t
  TlsSliceAfter(cost)
  If rc<>0 : Break : EndIf
  p=Service() : If p<0 : rc=p : Break : EndIf
 ForEver
 If rc=1
  n=Tls13ContextRestore(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
  If connection\tlsOp\result=1 : ProcedureReturn 0 : EndIf
  ProcedureReturn connection\tlsOp\result
 EndIf
 If connection\tlsOp\phase=1 : QuicTlsOperationCancel(@connection\tlsOp) : EndIf
 ProcedureReturn rc
EndProcedure
Procedure.i StartHandshake()
 Define rc.i,n.i,t.i,cost.i,pending.i
 rc=PrepareTlsProbe(1,0) : If rc<>1 : ProcedureReturn rc : EndIf
 rc=QuicConnectionStartBegin(@connection,20000000) : If rc<>0 : ProcedureReturn rc : EndIf
 If QuicConnectionStartBegin(@connection,20000000)<>#QUIC_START_STATE : startupFault=1 : EndIf
 If QuicConnectionSendBegin(@connection,0,0)<>#QUIC_SEND_STATE : startupFault=2 : EndIf
 If QuicConnectionReceiveBegin(@connection,0,@packet[0],1)<>#QUIC_RECEIVE_STATE : startupFault=3 : EndIf
 Repeat
  If connection\startup\phase=6 And operationProbes=1
   n=Tls13ContextRestore(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
   rc=PrepareTlsProbe(2,@connection\initial) : If rc<>1 : ProcedureReturn rc : EndIf
  EndIf
  pending=Bool(connection\startup\phase=2 Or connection\startup\phase=5 Or connection\startup\phase=7)
  If pending : TlsSliceBefore() : EndIf
  t=HwMicros() : rc=QuicConnectionStartStep(@connection) : cost=HwMicros()-t
  If pending : TlsSliceAfter(cost) : EndIf
  startupSteps=startupSteps+1 : If cost>maxStartupStep : maxStartupStep=cost : EndIf
  n=Service() : If n<0 : ProcedureReturn n : EndIf
  If rc<>0 : Break : EndIf
 ForEver
 If rc<>1 : ProcedureReturn rc : EndIf
 If QuicConnectionStartStep(@connection)<>1 Or QuicConnectionStartBegin(@connection,20000000)<>#QUIC_START_STATE : startupFault=4 : EndIf
 If connection\initial\readKeys\ready<>1 Or connection\initial\writeKeys\ready<>1 Or connection\initial\tx\tail<=0 : startupFault=5 : EndIf
 startupChecks=1
 n=Tls13ContextRestore(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
 ProcedureReturn 0
EndProcedure
Procedure.i SendSpace(*s.QuicConnectionSpace)
 Define rc.i,n.i,t.i,cost.i,held.i,i.i,number.i,inFlight.i
 rc=QuicConnectionSendBegin(@connection,*s\level,peerBytes)
 If rc<>0 : ProcedureReturn rc : EndIf
 Repeat
  n=Pump() : If n<0 : ProcedureReturn n : EndIf
  t=clock() : rc=QuicConnectionSendStep(@connection) : cost=clock()-t
  If cost>maxSendStep : maxSendStep=cost : EndIf
  sendSteps=sendSteps+1
  If connection\outgoing\worker\phase=1
   If TlsWorkerBegin(@otherJob,@CloseHoldEntry,0)<>#TLSWORK_BUSY : sendFault=1 : EndIf
  EndIf
  If rc=2
   If held=0
    number=connection\outgoing\number : inFlight=*s\history\inFlight
    For i=0 To 1199 : probeOutput[i]=output[i] : Next
    If QuicConnectionSendBegin(@connection,*s\level,peerBytes)<>#QUIC_SEND_STATE : sendFault=2 : EndIf
    If QuicConnectionSendAccepted(@connection,1199,clock())<>#QUIC_SEND_ARGUMENT : sendFault=3 : EndIf
    If QuicConnectionSendCompleted(@connection,0)<>#QUIC_SEND_STATE : sendFault=4 : EndIf
    If QuicConnectionSendStep(@connection)<>2 Or connection\outgoing\number<>number Or *s\history\inFlight<>inFlight : sendFault=5 : EndIf
    For i=0 To 1199 : If probeOutput[i]<>output[i] : sendFault=6 : EndIf : Next
    held=1 : sendProbes=sendProbes+1
   Else
    n=UdpPayloadSendTo(handle,@ep,QuicConnectionBuffer(@connection,16),connection\outgoing\bytes)
    If n<>#SVC_EAGAIN
     If n<1 : ProcedureReturn n : EndIf
     ; UDP returns an acceptance ticket, not a byte count. A positive ticket
     ; owns the whole datagram; acknowledge its requested length separately.
     n=QuicConnectionSendAccepted(@connection,connection\outgoing\bytes,clock())
     If n<>0 : ProcedureReturn n : EndIf
    EndIf
   EndIf
  ElseIf rc=3
   n=UdpPayloadStatus(handle,@status) : If n<>0 : ProcedureReturn n : EndIf
   If status\txState=#UDP_TX_SENT
    rc=QuicConnectionSendCompleted(@connection,0)
    If rc=1 : sent=sent+1 : initialDiscarded=connection\initialRetired : EndIf
   ElseIf status\txState=#UDP_TX_FAILED
    rc=QuicConnectionSendCompleted(@connection,status\txError)
   EndIf
  EndIf
  If rc<0 : ProcedureReturn rc : EndIf
 Until rc=1
 ProcedureReturn 0
EndProcedure
; Force an otherwise-due poll while a real ledger job holds its lock.
; This is a lifecycle refusal test, not a simulated wire timeout.
Procedure CheckRecoveryLocked(kind.i)
 Define before.QuicProbeTimer,i.i,rc.i,a.i,h.i,p.i
 a=connection\initial\history\busy : h=connection\handshake\history\busy : p=connection\applicationHistory\busy
 If a=0 And h=0 And p=0 : pollGuardFault=1 : ProcedureReturn : EndIf
 If a : pollSpaceMask=pollSpaceMask|1 : EndIf
 If h : pollSpaceMask=pollSpaceMask|2 : EndIf
 If p : pollSpaceMask=pollSpaceMask|4 : EndIf
 For i=0 To SizeOf(QuicProbeTimer)-1 : PokeA(@before+i,PeekA(@connection\recoveryTimer+i)) : Next
 rc=QuicConnectionRecoveryPoll(@connection,$7FFFFFFFFFFFFFFF)
 If rc<>#QUIC_SENT_BUSY : pollGuardFault=2 : EndIf
 For i=0 To SizeOf(QuicProbeTimer)-1
  If PeekA(@before+i)<>PeekA(@connection\recoveryTimer+i) : pollGuardFault=3 : EndIf
 Next
 If a<>connection\initial\history\busy Or h<>connection\handshake\history\busy Or p<>connection\applicationHistory\busy : pollGuardFault=4 : EndIf
 If kind=1 : pollAckChecks=pollAckChecks+1 : Else : pollLossChecks=pollLossChecks+1 : EndIf
EndProcedure
Procedure.i HandleAck(*s.QuicConnectionSpace)
 Define rc.i,n.i,phase.i,t.i,cost.i,authenticated.i
 phase=connection\acknowledge\phase
 authenticated=connection\receive\authenticated : connection\receive\authenticated=0
 rc=QuicConnectionAckBegin(@connection)
 If rc<>#QUIC_CONNECTION_ACK_STATE Or connection\acknowledge\phase<>phase Or *s\history\busy<>0 : ackFault=1 : EndIf
 connection\receive\authenticated=authenticated
 If authenticated<>1 : ackFault=7 : EndIf
 rc=QuicConnectionAckBegin(@connection)
 If rc<>0 : ProcedureReturn rc : EndIf
 If QuicConnectionAckBegin(@connection)<>#QUIC_CONNECTION_ACK_STATE : ackFault=2 : EndIf
 If QuicConnectionSendBegin(@connection,*s\level,peerBytes)<>#QUIC_SENT_BUSY : ackFault=3 : EndIf
 Repeat
  n=Pump() : If n<0 : ProcedureReturn n : EndIf
  If *s\history\busy<>0 : CheckRecoveryLocked(1) : EndIf
  t=clock() : rc=QuicConnectionAckStep(@connection) : cost=clock()-t
  ackSteps=ackSteps+1 : If cost>maxAckStep : maxAckStep=cost : EndIf
  If rc=0 And connection\receive\process\phase<>2 : ackFault=4 : EndIf
 Until rc<>0
 If rc<>1 : ProcedureReturn rc : EndIf
 If connection\receive\process\phase<>1 Or *s\history\busy<>0 : ackFault=5 : EndIf
 If QuicConnectionAckStep(@connection)<>1 Or QuicConnectionAckBegin(@connection)<>#QUIC_CONNECTION_ACK_STATE : ackFault=6 : EndIf
 ackEvents=ackEvents+connection\acknowledge\events : ackChecks=ackChecks+1
 ProcedureReturn 0
EndProcedure

 If connection\applicationState\ack\pending<>0 Or connection\applicationHistory\issued<>expected+1 Or connection\applicationHistory\first<>1 Or connection\applicationHistory\inFlight<>expected*probeBytes : probeFault=7 : EndIf

EndProcedure
Procedure.i ConfirmHandshake()
 Define rc.i,n.i,i.i,expected.i,beforeRead.QuicKeys,beforeWrite.QuicKeys
 CheckHandshakeSecretRetirement()
 For i=0 To #TLS13_CONTEXT_SIZE-1 : retrySnapshot[i]=ownerSnapshot[i] : Next
 For i=0 To SizeOf(QuicKeys)-1
  PokeA(@beforeRead+i,PeekA(@connection\appRead+i))
  PokeA(@beforeWrite+i,PeekA(@connection\appWrite+i))
 Next
 confirmForeign=Bool(peerBytes=0)
 If confirmForeign : ForeignContext() : EndIf
 n=Tls13ContextSave(@beforeSnapshot[0],#TLS13_CONTEXT_SIZE)
 rc=TlsWorkerBegin(@closeHold,@CloseHoldEntry,0)
 If rc<>1 : confirmFault=3 : ProcedureReturn rc : EndIf
 If QuicConnectionHandshakeDone(@connection)<>#TLSWORK_BUSY Or connection\handshakeConfirmed<>0 : confirmFault=4 : EndIf
 If TlsWorkerCancel(@closeHold)<>1 : confirmFault=5 : EndIf
 rc=QuicConnectionHandshakeDone(@connection)
 If rc<>1 : confirmFault=6 : ProcedureReturn rc : EndIf
 n=Tls13ContextSave(@afterSnapshot[0],#TLS13_CONTEXT_SIZE)
 For i=0 To #TLS13_CONTEXT_SIZE-1
  expected=retrySnapshot[i]
  If (i>=272 And i<=367) Or (i>=400 And i<=463) Or (i>=528 And i<=623)
   If expected<>0 : confirmSecretPresent=confirmSecretPresent+1 : EndIf
   expected=0
  EndIf
  If ownerSnapshot[i]<>expected : confirmFault=7 : EndIf
  If confirmForeign
   If beforeSnapshot[i]<>afterSnapshot[i] : confirmFault=8 : EndIf
  Else
   If ownerSnapshot[i]<>afterSnapshot[i] : confirmFault=9 : EndIf
  EndIf
 Next
 For i=0 To SizeOf(QuicKeys)-1
  If PeekA(@connection\handshake\readKeys+i)<>0 Or PeekA(@connection\handshake\writeKeys+i)<>0 : confirmFault=10 : EndIf
  If PeekA(@connection\appRead+i)<>PeekA(@beforeRead+i) Or PeekA(@connection\appWrite+i)<>PeekA(@beforeWrite+i) : confirmFault=11 : EndIf
 Next
 If confirmSecretPresent=0 : confirmFault=12 : EndIf
 If connection\handshakeConfirmed<>1 Or connection\timeout<>0 Or connection\handshake\history\phase<>2 Or connection\handshake\history\inFlight<>0 Or connection\handshake\rx\phase<>3 Or connection\handshake\tx\phase<>3 : confirmFault=13 : EndIf
 If QuicConnectionHandshakeDone(@connection)<>1 : confirmFault=14 : EndIf
 If QuicConnectionReceiveBegin(@connection,2,@packet[0],1)<>6 : confirmFault=15 : EndIf
 If QuicConnectionSendBegin(@connection,2,peerBytes)<>#QUIC_SEND_STATE : confirmFault=16 : EndIf
 If QuicConnectionTlsBegin(@connection,2,peerBytes,20000000)<>#QUIC_FEED_STATE : confirmFault=17 : EndIf
 If QuicConnectionCheckDeadline(@connection,connection\started+20000001)<>0 : confirmFault=18 : EndIf
 If connection\applicationState\ack\count<>0 : confirmFault=19 : EndIf
 n=Tls13ContextRestore(@ownerSnapshot[0],#TLS13_CONTEXT_SIZE)
 n=Tls13ContextWipe(@retrySnapshot[0],#TLS13_CONTEXT_SIZE)
 confirmChecked=1 : ProcedureReturn 1
EndProcedure
Procedure.i PacketOne(p.i,bytes.i)
 Define rc.i,i.i,t.i,cost.i,old.i,known.i,count.i,cursor.i,n.i,receivedAt.i,eliciting.i,*s.QuicConnectionSpace
 receivedAt=HwMicros()
 rc=QuicPacketRead(@connection\header,p,bytes,8) : If rc<>1 : ProcedureReturn -9307 : EndIf
 If connection\header\kind=4
  If connection\header\destinationBytes<>8 : ProcedureReturn -9308 : EndIf
  For i=0 To 7 : If PeekA(p+connection\header\destinationOffset+i)<>clientId[i] : ProcedureReturn -9309 : EndIf : Next
  If Tls13State<>#TLS13_ST_CONNECTED : ProcedureReturn -9310 : EndIf
  If connection\peerIds\phase=0
   rc=QuicConnectionPeerIdsInit(@connection,2) : If rc<>0 : ProcedureReturn rc : EndIf

  PokeA(p+savedApplicationBytes-1,PeekA(p+savedApplicationBytes-1)!1)
  rc=ApplicationCrypt(p,savedApplicationBytes,8,connection\applicationState\largestAuthenticated,0)
  If rc<>#QUIC_CRYPTO_AUTH : applicationFault=5 : ProcedureReturn -9442 : EndIf

   old=savedApplication[i] : If i=savedApplicationBytes-1 : old=old!1 : EndIf

  Next

  If QuicConnectionHandshakeDone(@connection)<>#QUIC_CONNECTION_STATE Or connection\handshakeConfirmed<>0 : confirmFault=2 : EndIf
  applicationBadTag=1
  For i=0 To savedApplicationBytes-1 : PokeA(p+i,savedApplication[i]) : Next
  rc=ApplicationCrypt(p,savedApplicationBytes,8,connection\applicationState\largestAuthenticated,0) : If rc<>1 : ProcedureReturn rc : EndIf

  If old<>0 Or connection\applicationState\largestAuthenticated<>applicationNumber Or connection\applicationState\ack\count<>0 : applicationStateFault=3 : EndIf
  cursor=0
  While cursor<connection\application\crypto\plainBytes
   n=QuicApplicationFrameRead(p+connection\application\crypto\plainOffset+cursor,connection\application\crypto\plainBytes-cursor,@applicationFrame)
   If n<1 : ProcedureReturn n : EndIf
   cursor=cursor+n
   If applicationFrame\base\kind<>0 And applicationFrame\base\kind<>1 And applicationFrame\base\kind<>24 And applicationFrame\base\kind<>30 : ProcedureReturn -9471 : EndIf
   eliciting=eliciting | applicationFrame\base\eliciting
   If applicationFrame\base\kind<>0 : applicationFrames=applicationFrames+1 : EndIf
   If applicationFrame\base\kind=30
    doneSeen=1 : rc=ConfirmHandshake() : If rc<>1 : ProcedureReturn rc : EndIf
   EndIf
   If applicationFrame\base\kind=24
    rc=QuicConnectionPeerIdAdd(@connection,applicationFrame\sequence,applicationFrame\retire,applicationFrame\base\data,applicationFrame\idBytes,applicationFrame\resetToken)
    If rc<0 : ProcedureReturn rc : EndIf
    peerIdsFrames=peerIdsFrames+1
    old=QuicPeerIdsFind(@connection\peerIds,applicationFrame\sequence)
    If old<=0 : peerIdsFault=2 : ProcedureReturn -9461 : EndIf
    If PeekI(old+8)<>applicationFrame\idBytes Or PeekI(old+24)<>1 : peerIdsFault=3 : EndIf
    For i=0 To applicationFrame\idBytes-1 : If PeekA(old+32+i)<>PeekA(applicationFrame\base\data+i) : peerIdsFault=4 : EndIf : Next
    For i=0 To 15 : If PeekA(old+52+i)<>PeekA(applicationFrame\resetToken+i) : peerIdsFault=5 : EndIf : Next
    If QuicConnectionPeerIdAdd(@connection,applicationFrame\sequence,applicationFrame\retire,applicationFrame\base\data,applicationFrame\idBytes,applicationFrame\resetToken)<>0 : peerIdsFault=6 : EndIf
   EndIf
   rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  Wend
  rc=QuicApplicationProcessed(@connection\applicationState,applicationNumber,eliciting,receivedAt)
  If rc<>1 : ProcedureReturn rc : EndIf
  controlReceipt=applicationNumber
  appAuthenticated=1 : ProcedureReturn 0
 EndIf
 If connection\header\kind=0 : *s=@connection\initial
 ElseIf connection\header\kind=2 : *s=@connection\handshake
 Else : ProcedureReturn -9311 : EndIf
 If savedReceiveBytes=0 And *s\level=0
  savedReceiveBytes=connection\header\packetBytes
  For i=0 To savedReceiveBytes-1 : savedReceive[i]=PeekA(p+i) : Next
  For i=0 To 7 : savedOriginal[i]=originalId[i] : savedClient[i]=clientId[i] : Next
 EndIf
 known=connection\peerIdKnown : count=connection\peerIdBytes
 old=PeekA(p+connection\header\destinationOffset)
 PokeA(p+connection\header\destinationOffset,old!1)
 rc=QuicConnectionReceivePacketBegin(@connection,p,connection\header\packetBytes)
 PokeA(p+connection\header\destinationOffset,old)
 If rc<>6 Or connection\peerIdKnown<>known Or connection\peerIdBytes<>count : cidFault=1 : EndIf
 If known<>0 And count>0
  old=PeekA(p+connection\header\sourceOffset)
  PokeA(p+connection\header\sourceOffset,old!1)
  rc=QuicConnectionReceivePacketBegin(@connection,p,connection\header\packetBytes)
  PokeA(p+connection\header\sourceOffset,old)
  If rc<>6 Or connection\peerIdKnown<>known Or connection\peerIdBytes<>count : cidFault=2 : EndIf
 EndIf
 rc=QuicConnectionReceivePacketBegin(@connection,p,connection\header\packetBytes)
 If rc=6 : ProcedureReturn 0 : EndIf
 If rc<>0 : ProcedureReturn rc : EndIf
 If connection\peerIdKnown<>known Or connection\peerIdBytes<>count : cidFault=3 : EndIf
 cidChecks=cidChecks+1
 Repeat
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  t=clock() : rc=QuicConnectionReceiveStep(@connection) : cost=clock()-t
  receiveSteps=receiveSteps+1 : If cost>maxReceiveStep : maxReceiveStep=cost : EndIf
  If TlsWorkerOwns(@connection\incoming\worker)=1
   If QuicConnectionReceiveBegin(@connection,*s\level,p,connection\header\packetBytes)<>#QUIC_RECEIVE_STATE : receiveFault=1 : EndIf
   If QuicConnectionSendBegin(@connection,*s\level,peerBytes)<>#QUIC_SEND_STATE : receiveFault=2 : EndIf
  EndIf
  If rc=2 : rc=HandleAck(*s) : If rc=0 : rc=1 : EndIf : EndIf
 Until rc<>1
 If rc=6 : ProcedureReturn 0 : EndIf
 If rc<>4 : ProcedureReturn rc : EndIf
 If QuicConnectionReceiveStep(@connection)<>4 : receiveFault=3 : EndIf
 peerBytes=connection\peerIdBytes
 If connection\peerIdKnown<>1 Or peerBytes<>connection\header\sourceBytes : cidFault=4 : EndIf
 rc=FeedTls(*s) : If rc<>0 : ProcedureReturn rc : EndIf
 ProcedureReturn SendSpace(*s)
EndProcedure

; Continue the actual exchange after TX; consume authenticated peer ACKs.
Procedure.i ReceiveApplicationAck()
 Define peerExponent.i,peerDelay.i,beforeAcked.i,*parameter.QuicParameter
 Define rc.i,n.i,i.i,p.i,bytes.i,cursor.i,used.i,pn.i,seen.i,eliciting.i,receivedAt.i,untilTime.i
 peerExponent=3 : peerDelay=25000
 *parameter=QuicParametersFind(@connection\params,10)
 If *parameter : peerExponent=*parameter\value : EndIf
 *parameter=QuicParametersFind(@connection\params,11)
 If *parameter : peerDelay=*parameter\value*1000 : EndIf
 If lossPhase=0
  rc=CheckDeliveryRefusals() : If rc<>0 : ProcedureReturn rc : EndIf
 EndIf
 untilTime=HwMicros()+3000000
 While (lossPhase=0 And applicationDelivery=0) Or (lossPhase=1 And lossAckCount<3-2*#PTO_PROBE_TEST)
  n=Service() : If n<0 : ProcedureReturn n : EndIf
  If HwMicros()>untilTime : ProcedureReturn -9481 : EndIf
  bytes=UdpPayloadReceive(handle,@info,@packet[0],1472)
  If bytes=#SVC_EAGAIN : Continue : EndIf
  If bytes<1 : ProcedureReturn bytes : EndIf
  If info\family<>4 Or info\peerPort<>55442 : ProcedureReturn -9482 : EndIf
  For i=0 To 15 : If PeekA(@info\peerAddress+i)<>PeekA(@ep\address+i) : ProcedureReturn -9483 : EndIf : Next
  p=@packet[0] : receivedAt=HwMicros()
  rc=QuicPacketRead(@connection\header,p,bytes,8) : If rc<>1 : Continue : EndIf
  If connection\header\kind<>4 : Continue : EndIf
  If connection\header\destinationBytes<>8 : ProcedureReturn -9484 : EndIf
  For i=0 To 7 : If PeekA(p+connection\header\destinationOffset+i)<>clientId[i] : ProcedureReturn -9485 : EndIf : Next
  rc=ApplicationCrypt(p,connection\header\packetBytes,8,connection\applicationState\largestAuthenticated,0)
  If rc<>1 : ProcedureReturn rc : EndIf
  pn=connection\application\crypto\number
  seen=QuicApplicationAuthenticated(@connection\applicationState,pn)
  If seen<0 : ProcedureReturn seen : EndIf
  If seen=1 : Continue : EndIf
  cursor=0 : eliciting=0
  While cursor<connection\application\crypto\plainBytes
   used=QuicApplicationFrameRead(p+connection\application\crypto\plainOffset+cursor,connection\application\crypto\plainBytes-cursor,@applicationFrame)
   If used<1 : ProcedureReturn used : EndIf
   cursor=cursor+used : eliciting=eliciting | applicationFrame\base\eliciting
   If applicationFrame\base\kind=0 Or applicationFrame\base\kind=1
    ElseIf applicationFrame\base\kind=2 Or applicationFrame\base\kind=3
     beforeAcked=connection\applicationHistory\acked
     rc=QuicHandshakeSentAckBegin(@connection\applicationAck,@connection\applicationHistory,4,@applicationFrame\base)
     If rc<>0 : ProcedureReturn rc : EndIf
     Repeat
      CheckRecoveryLocked(1)
 rc=QuicHandshakeSentAckStep(@connection\applicationAck) : deliverySteps=deliverySteps+1
      If rc=1
       If lossPhase=1
        lossAckCount=lossAckCount+1
        If connection\applicationAck\number<2 Or connection\applicationAck\number>4 Or connection\applicationAck\packetBytes<>probeBytes Or connection\applicationAck\inFlightBytes<>probeBytes : lossFault=1 : EndIf
       Else
       deliveryEvents=deliveryEvents+1
       If connection\applicationAck\number<>0 Or connection\applicationAck\packetBytes<>connection\controlBytes Or connection\applicationAck\inFlightBytes<>connection\controlBytes : deliveryFault=1 : EndIf
       EndIf
      EndIf
      n=Service() : If n<0 : ProcedureReturn n : EndIf
     Until rc=2 Or rc<0
     If rc<>2 : ProcedureReturn rc : EndIf
     rc=QuicRttAck(@connection\recoveryRtt,@connection\applicationAck,receivedAt,peerExponent,peerDelay,connection\handshakeConfirmed)
     If rc<0 : ProcedureReturn rc : EndIf
     If rc=1
      If rttSamples<4
       rttTrace[rttSamples*8]=connection\applicationAck\largest
       rttTrace[rttSamples*8+1]=receivedAt : rttTrace[rttSamples*8+2]=connection\applicationAck\rttSentAt
       rttTrace[rttSamples*8+3]=connection\applicationAck\rttDelay
       rttTrace[rttSamples*8+4]=peerExponent : rttTrace[rttSamples*8+5]=peerDelay
       rttTrace[rttSamples*8+6]=4 : rttTrace[rttSamples*8+7]=connection\handshakeConfirmed
      Else
       rttFault=1
      EndIf
      rttSamples=rttSamples+1
     EndIf
     rc=QuicConnectionRecoveryRearm(@connection,Bool(connection\applicationHistory\acked>beforeAcked And connection\handshakeConfirmed=1))
     If rc<0 : ProcedureReturn rc : EndIf
     If lossPhase=1 : ackRearmDue=connection\recoveryTimer\deadline : EndIf
     If lossPhase=0 And connection\applicationHistory\acked=1
      If connection\applicationHistory\inFlight<>0 Or connection\applicationHistory\largestAcked<>0 : deliveryFault=2 : EndIf
      applicationDelivery=1 : deliveryNumber=pn
      rc=QuicHandshakeSentPrune(@connection\applicationHistory,64)
      If rc<>1 Or connection\applicationHistory\first<>1 Or connection\applicationHistory\issued<>1 : deliveryFault=7 : EndIf
     EndIf
    ElseIf applicationFrame\base\kind=24
     rc=QuicConnectionPeerIdAdd(@connection,applicationFrame\sequence,applicationFrame\retire,applicationFrame\base\data,applicationFrame\idBytes,applicationFrame\resetToken)
     If rc<0 : ProcedureReturn rc : EndIf
    ElseIf applicationFrame\base\kind=30
     rc=QuicConnectionHandshakeDone(@connection) : If rc<>1 : ProcedureReturn rc : EndIf
    Else
     ProcedureReturn -9486
   EndIf
  Wend
  rc=QuicApplicationProcessed(@connection\applicationState,pn,eliciting,receivedAt)
  If rc<>1 : ProcedureReturn rc : EndIf
 Wend
 ProcedureReturn 0
EndProcedure

; Real missing-datagram proof. Legacy mode uses three later deliveries; PTO
; modes wait for deadlines, with optional consecutive encrypted datagram drops.
Procedure.i CheckProbeLoss()
 Define rc.i,n.i,i.i,highest.i,delay.i
 highest=4-2*#PTO_PROBE_TEST+#REPEAT_PROBE_TEST
 If #PTO_PROBE_TEST
  timerBase=QuicHandshakeSentProbeTime(@connection\applicationHistory)
  If connection\recoveryTimer\armed<>1 Or connection\recoveryTimer\level<>4 : ProcedureReturn -9493 : EndIf
  timerDue=connection\recoveryTimer\deadline
  Repeat
   n=Service() : If n<0 : QuicProbeTimerDiscard(@connection\recoveryTimer) : ProcedureReturn n : EndIf
   timerFired=HwMicros() : rc=QuicConnectionRecoveryPoll(@connection,timerFired)
  Until rc<>0
  If rc<>1 : ProcedureReturn rc : EndIf
  timerCount=timerCount+1
  If connection\recoveryTimer\level<>4 Or connection\recoveryTimer\backoff<>1 Or timerFired<timerDue : timerFault=1 : EndIf
  If QuicConnectionRecoveryPoll(@connection,HwMicros())<>0 : timerFault=2 : EndIf
  If connection\applicationHistory\inFlight<>probeBytes Or connection\applicationHistory\acked<>1 : timerFault=3 : EndIf
 EndIf
 For i=2 To highest
  If #REPEAT_PROBE_TEST And i=3
   repeatBase=QuicHandshakeSentProbeTime(@connection\applicationHistory)
   repeatDue=connection\recoveryTimer\deadline
   If connection\recoveryTimer\armed<>1 Or connection\recoveryTimer\backoff<>1 : timerFault=6 : EndIf
   Repeat
    n=Service() : If n<0 : ProcedureReturn n : EndIf
    repeatFired=HwMicros() : rc=QuicConnectionRecoveryPoll(@connection,repeatFired)
   Until rc<>0
   If rc<>1 : ProcedureReturn rc : EndIf
   timerCount=timerCount+1
   If connection\recoveryTimer\backoff<>2 Or connection\recoveryTimer\level<>4 Or repeatFired<repeatDue : timerFault=7 : EndIf
   If connection\applicationHistory\inFlight<>2*probeBytes Or connection\applicationHistory\acked<>1 : timerFault=8 : EndIf
   If QuicConnectionRecoveryPoll(@connection,HwMicros())<>0 : timerFault=9 : EndIf
  EndIf
  rc=SendApplicationProbe(i) : If rc<>0 : ProcedureReturn rc : EndIf
 Next
 rc=QuicConnectionLossBegin(@connection,4,HwMicros(),$7FFFFFFFFFFFFFFF,3)
 If rc<>0 : ProcedureReturn rc : EndIf
 If connection\applicationHistory\busy<>0 : CheckRecoveryLocked(2) : EndIf
 rc=QuicConnectionLossStep(@connection)
 If rc<>2 Or connection\applicationHistory\inFlight<>highest*probeBytes : lossFault=2 : EndIf
 If connection\recoveryLoss\phase=1 : QuicHandshakeSentLossCancel(@connection\recoveryLoss) : ProcedureReturn -9492 : EndIf
 lossPhase=1
 rc=ReceiveApplicationAck() : If rc<>0 : ProcedureReturn rc : EndIf
 If lossAckCount<>highest-1-#REPEAT_PROBE_TEST Or connection\applicationHistory\acked<>highest-#REPEAT_PROBE_TEST Or connection\applicationHistory\largestAcked<>highest Or connection\applicationHistory\inFlight<>(1+#REPEAT_PROBE_TEST)*probeBytes : lossFault=3 : EndIf
 delay=$7FFFFFFFFFFFFFFF
 If #PTO_PROBE_TEST
  delay=QuicRttLossDelay(@connection\recoveryRtt)
  If connection\recoveryTimer\backoff<>0 : timerFault=4 : EndIf
 EndIf
 rc=QuicConnectionLossBegin(@connection,4,HwMicros(),delay,3)
 If rc<>0 : ProcedureReturn rc : EndIf
 Repeat
  If connection\applicationHistory\busy<>0 : CheckRecoveryLocked(2) : EndIf
 rc=QuicConnectionLossStep(@connection)
  If rc=1
   lossEvents=lossEvents+1
   If connection\recoveryLoss\number<>lossEvents Or connection\recoveryLoss\packetBytes<>probeBytes Or connection\recoveryLoss\inFlightBytes<>probeBytes Or connection\recoveryLoss\sentAt<=0 : lossFault=4 : EndIf
  EndIf
  n=Service()
  If n<0
   If connection\recoveryLoss\phase=1 : QuicHandshakeSentLossCancel(@connection\recoveryLoss) : EndIf
   ProcedureReturn n
  EndIf
 Until rc=2 Or rc<0
 If rc<>2 : ProcedureReturn rc : EndIf
 If lossEvents<>1+#REPEAT_PROBE_TEST Or connection\applicationHistory\inFlight<>0 Or connection\applicationHistory\busy<>0 : lossFault=5 : EndIf
 If #PTO_PROBE_TEST
  If connection\recoveryTimer\armed<>0 : timerFault=5 : EndIf
 EndIf
 rc=QuicHandshakeSentPrune(@connection\applicationHistory,64)
 If rc<>0 Or connection\applicationHistory\first<>1 Or connection\applicationHistory\issued<>highest+1 : lossFault=6 : EndIf
 rc=QuicConnectionLossBegin(@connection,4,HwMicros(),$7FFFFFFFFFFFFFFF,3)
 If rc<>0 : ProcedureReturn rc : EndIf
 If connection\applicationHistory\busy<>0 : CheckRecoveryLocked(2) : EndIf
 rc=QuicConnectionLossStep(@connection)
 If rc<>2 Or connection\applicationHistory\inFlight<>0 : lossFault=7 : EndIf
 If connection\recoveryLoss\phase=1 : QuicHandshakeSentLossCancel(@connection\recoveryLoss) : ProcedureReturn -9493 : EndIf
 lossChecked=1 : ProcedureReturn 0
EndProcedure

Procedure.i Run()
 Define rc.i,n.i,i.i,p.i,packetBytes.i
 stage=1
 X509ResetAnchors() : X509AddAnchorEC(1,?RootDn,#ROOT_DN_BYTES,23,?RootKey,65)
 Tls13SetHsBuffer(@tlsHs[0],32768) : Tls13SetRecvBuffer(@tlsRecv[0],16645)
 QuicCipherInvalidateHook=@Tls13RecInvalidateCipher
 rc=Tls13Init(?HostName,15,#TEST_UNIX) : If rc<>1 : ProcedureReturn -9316 : EndIf
 stage=2
 rc=DrbgAutoSeed() : If rc<>1 : ProcedureReturn rc : EndIf
 rc=DrbgGenerate(@originalId[0],8) : If rc<>8 : ProcedureReturn -9317 : EndIf
 rc=DrbgGenerate(@clientId[0],8) : If rc<>8 : ProcedureReturn -9318 : EndIf
 rc=Tls13SetAlpn(?Alpn,3) : If rc<>1 : ProcedureReturn -9320 : EndIf
 For i=0 To #LOCAL_PARAMETER_BYTES-1 : localParameters[i]=PeekA(?LocalTemplate+i) : Next
 For i=0 To 7 : localParameters[i+2]=clientId[i] : Next
 rc=Tls13SetQuicParameters(@tlsConfig[0],@localParameters[0],#LOCAL_PARAMETER_BYTES,@peerParameters[0],4096)
 If rc<>1 : ProcedureReturn -9321 : EndIf
 stage=3
 rc=StartHandshake() : If rc<>0 : ProcedureReturn rc : EndIf
 rc=SendSpace(@connection\initial) : If rc<>0 : ProcedureReturn rc : EndIf
 stage=4
 While appAuthenticated=0
  rc=Pump() : If rc<0 : ProcedureReturn rc : EndIf
  n=UdpPayloadReceive(handle,@info,@packet[0],1472)
  If n<>#SVC_EAGAIN
   If n<1 : ProcedureReturn n : EndIf
   If info\family<>4 Or info\peerPort<>55442 : ProcedureReturn -9322 : EndIf
   For i=0 To 15 : If PeekA(@info\peerAddress+i)<>PeekA(@ep\address+i) : ProcedureReturn -9323 : EndIf : Next
   received=received+1 : p=0
   While p<n
    ; An unparseable datagram remainder is discarded, not a connection error.
    ; Already authenticated coalesced packets retain their effects.
    rc=QuicPacketRead(@connection\header,@packet[p],n-p,8) : If rc<>1 : Break : EndIf
    packetBytes=connection\header\packetBytes
    rc=PacketOne(@packet[p],packetBytes) : If rc<>0 : ProcedureReturn rc : EndIf
    p=p+packetBytes
   Wend
  EndIf
 Wend
 stage=5 : rc=SendApplicationPing() : If rc<>0 : ProcedureReturn rc : EndIf
 rc=ReceiveApplicationAck() : If rc<>0 : ProcedureReturn rc : EndIf
 rc=SendApplicationProbe(1) : If rc<>0 : ProcedureReturn rc : EndIf
 If #DROP_PROBE_TEST : ProcedureReturn CheckProbeLoss() : EndIf
 ProcedureReturn 0
EndProcedure
Procedure.i LiveEntry()

  Case 1 : liveResult=Tls13NewKeyShare()
  Case 2 : liveResult=QuicTlsOutputBegin(@connection\tlsOutput,@*s\tx,@tlsBytes[0],2048)
  Case 3 : liveResult=QuicCryptoTlsStep(@*s\rx)
  Default : liveResult=-9325
 EndSelect

EndProcedure
; Compare complete snapshots, not just a few state flags. The foreground
