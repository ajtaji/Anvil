Procedure.i Capture(cmd.i,p.i,n.i,ms.i)
 Define i.i
 capturedBytes=n : capturedKind=cmd : capturedMs=ms
 If n>192 : ProcedureReturn -999 : EndIf
 For i=0 To n-1 : captured[i]=PeekA(p+i) : Next
 ProcedureReturn fakeRc
EndProcedure
Procedure.i Cyw43IoctlSet(cmd.i,p.i,n.i,ms.i)
 ProcedureReturn Capture(cmd,p,n,ms)
EndProcedure
Procedure.i Cyw43IoctlGet(cmd.i,p.i,n.i,ms.i)
 ProcedureReturn Capture(cmd,p,n,ms)
EndProcedure
Procedure cyw43_PutLE32(p.i, v.i)
  PokeB(p,     v & $FF)
  PokeB(p + 1, (v >> 8) & $FF)
  PokeB(p + 2, (v >> 16) & $FF)
  PokeB(p + 3, (v >> 24) & $FF)
EndProcedure
Procedure.i cyw43_PutName(dst.i, src.i, namelen.i)
  Define i.i
  For i = 0 To namelen - 1
    PokeB(dst + i, PeekA(src + i))
  Next
  PokeB(dst + namelen, 0)
  ProcedureReturn namelen + 1
EndProcedure
Procedure.i Cyw43ReplyPtr()
  ProcedureReturn @cyw43_rx[0] + cyw43_replyOff
EndProcedure
Procedure.i Cyw43ReplyLen()
  ProcedureReturn cyw43_replyLen
EndProcedure
Procedure.i Cyw43SetIovarU32(src.i, namelen.i, v.i, timeoutMs.i)
  Define n.i
  n = cyw43_PutName(@cyw43_arg[0], src, namelen)
  cyw43_PutLE32(@cyw43_arg[0] + n, v)
  ProcedureReturn Cyw43IoctlSet(#CYW43_C_SET_VAR, @cyw43_arg[0], n + 4, timeoutMs)
EndProcedure
Procedure.i Cyw43SetIovarData(src.i, namelen.i, *data, datalen.i, timeoutMs.i)
  Define n.i
  Define i.i
  If datalen < 0
    ProcedureReturn #CYW43_IO_TOOBIG
  EndIf
  n = cyw43_PutName(@cyw43_arg[0], src, namelen)
  If (n + datalen) > #CYW43_ARGBUF
    ProcedureReturn #CYW43_IO_TOOBIG
  EndIf
  For i = 0 To datalen - 1
    PokeB(@cyw43_arg[0] + n + i, PeekA(*data + i))
  Next
  ProcedureReturn Cyw43IoctlSet(#CYW43_C_SET_VAR, @cyw43_arg[0], n + datalen, timeoutMs)
EndProcedure
Procedure.i Cyw43GetIovarData(src.i, namelen.i, buflen.i, timeoutMs.i)
  Define n.i
  Define i.i
  If buflen > #CYW43_ARGBUF
    ProcedureReturn #CYW43_IO_TOOBIG
  EndIf
  n = cyw43_PutName(@cyw43_arg[0], src, namelen)
  If n > buflen
    ProcedureReturn #CYW43_IO_TOOBIG
  EndIf
  For i = n To buflen - 1
    PokeB(@cyw43_arg[0] + i, 0)
  Next
  ProcedureReturn Cyw43IoctlGet(#CYW43_C_GET_VAR, @cyw43_arg[0], buflen, timeoutMs)
EndProcedure
